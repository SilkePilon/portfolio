# Full-site CMS editing + live preview — implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** every text/image on the site is editable in Payload (`/admin`), the admin is tidy, and every document shows a live preview that updates while typing.

**Architecture:** Payload globals `home`, `pages`, `site` + collections `services`, `testimonials`, `clients`, `awards`, `faqs` (orderable) hold all copy. Pure client-safe mappers (`src/lib/*-map.ts`) turn Payload docs into the existing static content shapes (`typeof home`, `typeof pages`, `SiteContent`, `Service[]`…). A single `ContentProvider` React context exposes them to components via hooks; the server fetches everything once per request (`getContent()`); a `LivePreviewBridge` client component patches the context from Payload's `window.postMessage` stream.

**Tech Stack:** Next.js 16 App Router, Payload 3.88 (`@payloadcms/db-sqlite`, `@payloadcms/live-preview` 3.88), React 19, Vitest + Testing Library.

Spec: `docs/superpowers/specs/2026-08-26-full-cms-design.md` — read it first.

## Global constraints

- Static content in `src/content/*.ts` stays the fallback + seed source; every mapper falls back per field to it.
- Marked text: `**white**` markers, `\n` = line break (spec "Marked text").
- No drafts/versions. `orderable: true` replaces numeric `order` fields.
- Access: `publicRead` / `authenticated` from `src/cms/access.ts` on every new collection/global.
- Admin groups exactly: Pages · Content · Home lists · Inbox · Admin.
- Run `npm run generate:types` after any collection/global change; run `npm run payload -- migrate:create <name>` once at the end (Task 7) and commit `src/migrations/`.
- Tests: `npm test` must stay green after every task; `npm run typecheck` clean.
- Commit after every task with a conventional message ending in `Co-Authored-By: Claude Fable 5 <noreply@anthropic.com>`.
- Dev server may be running on :3000 (`npm run dev`); restart it if config changes don't show up. Never `pkill -f`; use `lsof -ti :3000 | xargs -r kill`.

```ts
/** Adjacent same-colour parts merge on parse, so compare against the merged form. */
function normalise(r: Rich): Rich {
  const out: Rich = []
  for (const p of r) {
    const last = out[out.length - 1]
    if (p.br) out.push({ br: true })
    else if (last && !last.br && !!last.muted === !!p.muted) last.text = (last.text ?? '') + (p.text ?? '')
    else out.push(p.muted ? { text: p.text ?? '', muted: true } : { text: p.text ?? '' })
  }
  return out
}
```

```ts
// src/lib/marked.ts
import type { Rich } from '@/content/types'

/**
 * Plain-text form of the mixed-colour headings used everywhere in the template:
 * text is grey by default, `**like this**` renders white, a newline is a line break.
 * Unbalanced `**` is kept literally. Adjacent parts of the same colour are merged.
 */
export function parseMarked(s: string): Rich {
  const out: Rich = []
  const push = (text: string, muted: boolean) => {
    if (!text) return
    const last = out[out.length - 1]
    if (last && !last.br && !!last.muted === muted) last.text = (last.text ?? '') + text
    else out.push(muted ? { text, muted: true } : { text })
  }
  for (const [li, line] of s.split(/\r?\n/).entries()) {
    if (li > 0) out.push({ br: true })
    const parts = line.split('**')
    if (parts.length % 2 === 0) { push(line, true); continue } // unbalanced → literal
    parts.forEach((p, i) => push(p, i % 2 === 0))
  }
  return out
}

/** Inverse of parseMarked (seed + tests). */
export function toMarked(r: Rich): string {
  return r.map((p) => (p.br ? '\n' : p.muted ? p.text ?? '' : p.text ? `**${p.text}**` : '')).join('')
}
```

```tsx
// src/components/layout/LivePreviewBridge.tsx
'use client'
import { useEffect } from 'react'
import { mergeData, ready } from '@payloadcms/live-preview'
import { applyPreview } from '@/lib/preview-patch'
import { useSetContent } from './ContentProvider'

/** Inside the Payload admin's live-preview iframe: patches the content context as the editor types. No-op elsewhere. */
export function LivePreviewBridge() {
  const setContent = useSetContent()
  useEffect(() => {
    if (window.parent === window) return
    const serverURL = window.location.origin
    const onMessage = async (event: MessageEvent) => {
      const d = event.data
      if (event.origin !== serverURL || !d || d.type !== 'payload-live-preview') return
      const { collectionSlug, globalSlug, data } = d
      if (!collectionSlug && !globalSlug) return
      try {
        // Ask Payload to populate uploads/relations of the unsaved form data.
        const doc = await mergeData({ apiRoute: '/api', collectionSlug, globalSlug, depth: 1, incomingData: data, initialData: { id: data?.id }, serverURL })
        setContent((c) => applyPreview(c, { collectionSlug, globalSlug, doc }))
      } catch (e) { console.warn('[live-preview]', e) }
    }
    window.addEventListener('message', onMessage)
    ready({ serverURL })
    return () => window.removeEventListener('message', onMessage)
  }, [setContent])
  return null
}
```
